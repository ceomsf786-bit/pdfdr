# Run these from inside the SNT PDF Board folder.
# Replace YOUR_PROJECT_REF with the short project reference from your Supabase URL/dashboard.

supabase login

supabase functions deploy viewer-api --no-verify-jwt

git init

git remote add origin https://github.com/YOUR_USERNAME/YOUR_REPOSITORY.git
git push -u origin main
